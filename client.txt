DEFAULT_UPLOAD_URL="http://192.168.0.101:5000/phone/api/upload/screenshot" # <<< Đã sửa URL dựa trên log của bạn

# Tên của trường dữ liệu file mà server Flask mong đợi
FILE_FIELD_NAME="screenshot_file"

# === ĐƯỜNG DẪN ĐẦY ĐỦ ĐẾN CURL TRONG TERMUX ===
CURL_EXEC="/data/data/com.termux/files/usr/bin/curl"

# === KIỂM TRA THAM SỐ ===
FILE_PATH=$1
UPLOAD_URL=${2:-$DEFAULT_UPLOAD_URL}

if [ -z "$FILE_PATH" ]; then
  echo "Lỗi: Vui lòng cung cấp đường dẫn đến file ảnh cần upload."
  echo "Cách dùng: $0 /đường/dẫn/tới/ảnh.png [URL_upload_tùy_chọn]"
  exit 1
fi

if [ ! -f "$FILE_PATH" ]; then
  echo "Lỗi: File không tồn tại tại đường dẫn: $FILE_PATH"
  exit 1
fi

# === THỰC HIỆN UPLOAD BẰNG CURL ===
echo "Đang upload file: $FILE_PATH"
echo "Đến URL: $UPLOAD_URL"
echo "Sử dụng trường file: $FILE_FIELD_NAME"

# Sử dụng biến $CURL_EXEC để gọi curl với đường dẫn đầy đủ
$CURL_EXEC -X POST \
     -F "$FILE_FIELD_NAME=@$FILE_PATH" \
     -sS \
     -w "\nHTTP Status: %{http_code}\n" \
     "$UPLOAD_URL"

if [ $? -eq 0 ]; then
  echo "Lệnh curl đã thực thi."
else
  echo "Lỗi khi thực thi lệnh curl. Mã lỗi: $?" # Thêm mã lỗi
  # Có thể thêm log chi tiết hơn ở đây nếu cần
  # $CURL_EXEC -X POST -F "$FILE_FIELD_NAME=@$FILE_PATH" -v "$UPLOAD_URL" # Thêm -v để debug
  exit 1
fi

echo "Quá trình upload hoàn tất."
exit 0
