{# templates/_flash_messages.html #} {# File này chịu trách nhiệm hiển thị các tin nhắn flash (thông báo nhanh) từ Flask #} {# Nó sử dụng hàm get_flashed_messages(with_categories=true) để lấy tin nhắn #} {% with messages = get_flashed_messages(with_categories=true) %} {# Chỉ hiển thị nếu có tin nhắn được flash #} {% if messages %} {# Lặp qua từng cặp (category, message) #} {% for category, message in messages %} {# Chuyển đổi category của Flask thành class alert của Bootstrap 5. Ví dụ: - flash('Thông báo thành công', 'success') -> alert-success (xanh lá) - flash('Đã xảy ra lỗi!', 'error') -> alert-danger (đỏ) - flash('Cảnh báo nhé.', 'warning') -> alert-warning (vàng) - flash('Thông tin thêm.') hoặc flash('Thông tin thêm.', 'info') -> alert-info (xanh dương nhạt) - Các category khác không khớp sẽ dùng alert-secondary (xám) làm mặc định. #} {% set alert_type = category | lower %} {% if alert_type == 'message' or alert_type == 'info' %} {% set alert_class = 'info' %} {% elif alert_type == 'error' %} {% set alert_class = 'danger' %} {% elif alert_type not in ['primary', 'secondary', 'success', 'danger', 'warning', 'light', 'dark'] %} {% set alert_class = 'secondary' %} {# Mặc định cho các category lạ #} {% else %} {% set alert_class = alert_type %} {# Giữ nguyên success, warning, etc. #} {% endif %} {# Tạo cấu trúc HTML cho Alert của Bootstrap 5 #} {# mb-3: tạo khoảng cách dưới nếu có nhiều thông báo #} {# alert-dismissible fade show: cho phép đóng và có hiệu ứng mờ dần #}
